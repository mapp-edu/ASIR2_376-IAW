
### 1.1 Configuración (local al repositorio para no tocar la del equipo)
$ git init -q
$ git config user.name "Ana Alumna"
$ git config user.email "ana.alumna@example.com"
$ git symbolic-ref HEAD refs/heads/main
$ git config --list --local
core.repositoryformatversion=0
core.filemode=true
core.bare=false
core.logallrefupdates=true
user.name=Ana Alumna
user.email=ana.alumna@example.com

### 1.3 README.md
$ git status --short
?? README.md
$ git add README.md
$ git status --short
A  README.md
$ git commit -q -m "Añade el README con el propósito del repositorio"

### 1.4 config/puertos.txt
$ git add config/puertos.txt

### 2.1 Borrar una línea sin preparar y recuperarla
$ git status --short
 M config/puertos.txt
$ git restore config/puertos.txt
$ git status --short

### 2.2 Borrar, preparar, sacar del área de preparación y recuperar
$ git add config/puertos.txt
$ git status --short
M  config/puertos.txt
$ git restore --staged config/puertos.txt
$ git status --short
 M config/puertos.txt
$ git restore config/puertos.txt
$ git status --short

### 2.3 Borrado accidental confirmado y deshecho con revert
$ git rm -q README.md
$ git commit -q -m "Elimina el README"
$ git revert --no-edit HEAD

 create mode 100644 README.md
$ ls

### 2.4 .gitignore
$ git status --short

$ git add .gitignore
$ git commit -q -m "Ignora los registros y el archivo de secretos"

!! secretos.txt
$ git check-ignore -v acceso.log secretos.txt
